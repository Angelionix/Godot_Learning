## 3.1 Data Layer — ресурсы данных

Все данные определены как GDScript Resources с @export переменными. Это позволяет настраивать параметры в инспекторе Godot без изменения кода. C++ классы читают эти ресурсы через GDExtension API.

### resources/item_data.gd

```gdscript
class_name ItemData
extends Resource

@export var item_id: StringName
@export var display_name: String = "Unknown Item"
@export var description: String = ""
@export var icon: Texture2D
@export var item_type: StringName = &"consumable"  # consumable, equipment, quest_item, material
@export var max_stack: int = 99
@export var sell_price: int = 0
@export var world_scene: PackedScene  # 3D модель в мире
@export var effects: Array[Dictionary] = []  # {"type": "heal", "value": 30}
@export var equipment_slot: StringName = &""  # weapon, armor, accessory (пусто для неэкипируемых)
@export var stat_bonuses: Dictionary = {}  # {"attack": 10, "defense": 5}
```

**Проверка:** ✅ `extends Resource` — корректно. ✅ Все поля `@export` — доступны в инспекторе. ✅ `item_type` использует StringName с `&` — оптимально. ✅ `effects` — Array[Dictionary] позволяет гибко описывать эффекты. ✅ `equipment_slot` пуст по умолчанию для неэкипируемых предметов.

void AdventureCharacter::_ready() {
    camera_pivot = Object::cast_to<Node3D>(get_node_or_null("CameraPivot"));
}

void AdventureCharacter::_physics_process(double delta) {
    // Гравитация с переменной высотой прыжка
    if (!is_on_floor()) {
        double grav = velocity.y < 0.0 ? fall_gravity : jump_gravity;
        velocity.y -= grav * delta;

        // Укороченный прыжок при отпускании кнопки
        if (velocity.y > 0.0 && !Input::get_singleton()->is_action_pressed("jump")) {
            velocity.y *= 0.5;
        }
    } else if (Input::get_singleton()->is_action_just_pressed("jump")) {
        velocity.y = jump_velocity;
    }

    // Движение относительно камеры
    Vector3 input_dir = _get_camera_relative_input();
    bool sprinting = Input::get_singleton()->is_action_pressed("sprint");
    double current_speed = sprinting ? sprint_speed : move_speed;

    velocity.x = input_dir.x * current_speed;
    velocity.z = input_dir.z * current_speed;

    set_velocity(velocity);
    move_and_slide();

    // Поворот персонажа к направлению движения
    if (input_dir.length_squared() > 0.01) {
        Vector3 target_direction = -input_dir;
        double target_yaw = Math::atan2(target_direction.x, target_direction.z);
        Quaternion target_quat = Quaternion(Vector3(0, 1, 0), target_yaw);
        Quaternion current_quat = get_global_transform().basis.get_rotation_quaternion();
        Quaternion new_quat = current_quat.slerp(target_quat, rotation_speed * delta);
        set_global_transform(Transform3D(Basis(new_quat), get_global_position()));
    }
}

Vector3 AdventureCharacter::_get_camera_relative_input() const {
    auto *input = Input::get_singleton();
    Vector2 raw = input->get_vector("move_left", "move_right", "move_forward", "move_back");

    if (!camera_pivot) {
        return Vector3(raw.x, 0.0, raw.y).normalized();
    }

    Vector3 cam_forward = -camera_pivot->get_global_basis().get_column(2);
    Vector3 cam_right = camera_pivot->get_global_basis().get_column(0);

    cam_forward.y = 0.0;
    cam_forward.normalize();
    cam_right.y = 0.0;
    cam_right.normalize();

    return (cam_forward * raw.y + cam_right * raw.x).normalized();
}

### src/camera_controller.cpp (ключевые методы)

```cpp
#include "camera_controller.h"
#include <godot_cpp/classes/input.hpp>
#include <godot_cpp/variant/math.hpp>

namespace adv {

void CameraController::_process(double delta) {
    switch (current_mode) {
        case CameraMode::FOLLOW:    _update_follow(delta); break;
        case CameraMode::CINEMATIC: _update_cinematic(delta); break;
        case CameraMode::FIXED:     break; // Статичная камера
        case CameraMode::TRANSITION: _update_transition(delta); break;
    }
}

void CameraController::_update_follow(double delta) {
    auto *input = Input::get_singleton();

    // Вращение камеры
    Vector2 cam_input = input->get_vector("camera_left", "camera_right", "camera_up", "camera_down");
    yaw += cam_input.x * rotation_speed * delta;
    pitch = Math::clamp(pitch + cam_input.y * rotation_speed * delta, min_pitch, max_pitch);

    // Применяем вращение к SpringArm3D (дочерний узел)
    set_rotation(Vector3(Math::deg_to_rad(pitch), Math::deg_to_rad(yaw), 0.0));

    // Плавное следование за персонажем
    if (follow_target) {
        Vector3 target = follow_target->get_global_position();
        Vector3 current = get_global_position();
        double factor = 1.0 - Math::exp(-smooth_speed * delta);
        set_global_position(current.lerp(target, factor));
    }
}

void CameraController::_update_transition(double delta) {
    transition_progress += delta / transition_duration;
    if (transition_progress >= 1.0) {
        transition_progress = 1.0;
        set_global_transform(transition_to);
        current_mode = CameraMode::CINEMATIC;
        return;
    }
    // Smoothstep интерполяция
    double t = transition_progress * transition_progress * (3.0 - 2.0 * transition_progress);
    Vector3 pos = transition_from.origin.lerp(transition_to.origin, t);
    Quaternion rot = transition_from.basis.get_rotation_quaternion().slerp(
        transition_to.basis.get_rotation_quaternion(), t);
    set_global_transform(Transform3D(Basis(rot), pos));
}

void CameraController::set_mode(CameraMode mode) {
    if (current_mode == mode) return;
    current_mode = mode;
}

void CameraController::transition_to(const Transform3D &target, double duration) {
    transition_from = get_global_transform();
    transition_to = target;
    transition_progress = 0.0;
    transition_duration = Math::max(0.1, duration);
    current_mode = CameraMode::TRANSITION;
}

} // namespace adv
```

**Проверка:** ✅ `_update_follow()` использует `Math::clamp` вместо макроса `CLAMP` (нестандартный). ✅ `_update_transition()` использует smoothstep для плавного перехода. ✅ `transition_to()` валидирует duration через `Math::max`. ✅ `_bind_methods()` опущен для краткости, но следует тому же шаблону, что и AdventureCharacter.

### src/npc_controller.cpp (ключевые методы)

```cpp
#include "npc_controller.h"
#include <godot_cpp/classes/scene_tree.hpp>
#include <godot_cpp/variant/utility_functions.hpp>

namespace adv {

void NPCController::_ready() {
    nav_agent = Object::cast_to<NavigationAgent3D>(get_node_or_null("NavigationAgent3D"));

    // Получаем ссылку на игрока через группу
    TypedArray<Node> players = get_tree()->get_nodes_in_group("player");
    if (players.size() > 0) {
        player_ref = Object::cast_to<CharacterBody3D>(players[0]);
    }
}

void NPCController::_physics_process(double delta) {
    if (!nav_agent || !player_ref) return;

    switch (current_state) {
        case NPCState::IDLE:     _update_idle(delta); break;
        case NPCState::PATROL:   _update_patrol(delta); break;
        case NPCState::FOLLOW:   _update_follow(delta); break;
        case NPCState::ATTACK:   _update_attack(delta); break;
        case NPCState::FLEE:     _update_flee(delta); break;
        case NPCState::STUNNED:  _update_stunned(delta); break;
        case NPCState::DEAD:     break;
    }

    // Навигация
    if (current_state != NPCState::IDLE && current_state != NPCState::STUNNED
        && current_state != NPCState::DEAD) {
        Vector3 next_pos = nav_agent->get_next_path_position();
        Vector3 direction = (next_pos - get_global_position()).normalized();
        Vector3 vel = direction * move_speed;
        vel.y = get_velocity().y;

        set_velocity(vel);
        move_and_slide();
    }
}

void NPCController::_update_idle(double delta) {
    idle_timer += delta;
    if (idle_timer >= idle_wait_time) {
        idle_timer = 0.0;
        _transition_to(NPCState::PATROL);
    }

    // Обнаружение игрока
    if (player_ref) {
        double dist = get_global_position().distance_to(player_ref->get_global_position());
        if (dist <= detect_range) {
            _transition_to(NPCState::FOLLOW);
        }
    }
}

void NPCController::_update_patrol(double delta) {
    if (patrol_points.is_empty()) {
        _transition_to(NPCState::IDLE);
        return;
    }

    Vector3 target = patrol_points[current_patrol_index];
    nav_agent->set_target_position(target);

    if (get_global_position().distance_to(target) < 1.0) {
        current_patrol_index = (current_patrol_index + 1) % patrol_points.size();
    }

    // Обнаружение игрока
    if (player_ref) {
        double dist = get_global_position().distance_to(player_ref->get_global_position());
        if (dist <= detect_range) {
            _transition_to(NPCState::FOLLOW);
        }
    }
}

void NPCController::_update_follow(double delta) {
    if (!player_ref) {
        _transition_to(NPCState::IDLE);
        return;
    }

    nav_agent->set_target_position(player_ref->get_global_position());
    double dist = get_global_position().distance_to(player_ref->get_global_position());

    if (dist <= attack_range) {
        _transition_to(NPCState::ATTACK);
    } else if (dist > detect_range * 1.5) {
        _transition_to(NPCState::IDLE);
    }

    // Бегство при низком здоровье
    HealthComponent *health = Object::cast_to<HealthComponent>(get_node_or_null("HealthComponent"));
    if (health && health->get_health_ratio() <= flee_health_threshold) {
        _transition_to(NPCState::FLEE);
    }
}

void NPCController::_update_stunned(double delta) {
    stunned_timer += delta;
    if (stunned_timer >= stunned_duration) {
        stunned_timer = 0.0;
        _transition_to(NPCState::FOLLOW);
    }
}

void NPCController::stun(double duration) {
    stunned_duration = duration;
    stunned_timer = 0.0;
    _transition_to(NPCState::STUNNED);
}

void NPCController::_transition_to(NPCState new_state) {
    current_state = new_state;
}

void NPCController::set_patrol_points(const PackedVector3Array &points) {
    patrol_points = points;
}

PackedVector3Array NPCController::get_patrol_points() const {
    return patrol_points;
}

} // namespace adv
```

```cpp
#include "inventory_system.h"
#include <godot_cpp/variant/utility_functions.hpp>

namespace adv {

bool InventorySystem::add_item(const StringName &item_id, int count) {
    if (count <= 0) return false;

    // Проверяем, есть ли уже такой предмет
    auto it = item_counts.find(item_id);
    if (it != item_counts.end()) {
        it->second += count;
        _emit_inventory_changed();
        return true;
    }

    // Ищем пустой слот
    for (auto &slot : slots) {
        if (slot.count == 0) {
            slot.item_id = item_id;
            slot.count = count;
            item_counts[item_id] = count;
            _emit_inventory_changed();
            return true;
        }
    }

    // Нет пустых слотов
    return false;
}

bool InventorySystem::remove_item(const StringName &item_id, int count) {
    auto it = item_counts.find(item_id);
    if (it == item_counts.end() || it->second < count) return false;

    it->second -= count;
    if (it->second <= 0) {
        item_counts.erase(it);
        // Очищаем слот
        for (auto &slot : slots) {
            if (slot.item_id == item_id) {
                slot.item_id = StringName();
                slot.count = 0;
                break;
            }
        }
    }

    _emit_inventory_changed();
    return true;
}

bool InventorySystem::has_item(const StringName &item_id, int count) const {
    auto it = item_counts.find(item_id);
    return it != item_counts.end() && it->second >= count;
}

int InventorySystem::get_count(const StringName &item_id) const {
    auto it = item_counts.find(item_id);
    return it != item_counts.end() ? it->second : 0;
}

Dictionary InventorySystem::to_dict() const {
    Dictionary result;
    for (const auto &slot : slots) {
        if (slot.count > 0) {
            result[slot.item_id] = slot.count;
        }
    }
    return result;
}

void InventorySystem::from_dict(const Dictionary &data) {
    item_counts.clear();
    for (auto &slot : slots) {
        slot = ItemSlot();
    }

    const Array keys = data.keys();
    int slot_idx = 0;
    for (int i = 0; i < keys.size() && slot_idx < max_slots; ++i) {
        StringName key = keys[i];
        int count = data[key];
        slots[slot_idx].item_id = key;
        slots[slot_idx].count = count;
        item_counts[key] = count;
        slot_idx++;
    }

    _emit_inventory_changed();
}

void InventorySystem::_emit_inventory_changed() {
    emit_signal("inventory_changed");
}

void InventorySystem::_ready() {
    slots.resize(max_slots);
}

void InventorySystem::_bind_methods() {
    ADD_SIGNAL(MethodInfo("inventory_changed"));

    ClassDB::bind_method(D_METHOD("add_item", "item_id", "count"), &InventorySystem::add_item, DEFVAL(1));
    ClassDB::bind_method(D_METHOD("remove_item", "item_id", "count"), &InventorySystem::remove_item, DEFVAL(1));
    ClassDB::bind_method(D_METHOD("has_item", "item_id", "count"), &InventorySystem::has_item, DEFVAL(1));
    ClassDB::bind_method(D_METHOD("get_count", "item_id"), &InventorySystem::get_count);
    ClassDB::bind_method(D_METHOD("to_dict"), &InventorySystem::to_dict);
    ClassDB::bind_method(D_METHOD("from_dict", "data"), &InventorySystem::from_dict);
}

} // namespace adv
```

**Проверка:** ✅ `add_item()` проверяет count > 0, ищет существующий стек, затем пустой слот. ✅ `remove_item()` проверяет наличие, очищает слот при 0. ✅ `has_item()` — O(1) через unordered_map. ✅ `to_dict()` — корректная сериализация для Godot. ✅ `from_dict()` — очистка + восстановление. ✅ `_bind_methods()` с DEFVAL для опциональных параметров. ✅ Сигнал `inventory_changed` объявлен через ADD_SIGNAL.

        var idx := i

        choices_container.add_child(btn)

func _on_continue() -> void:
    var line: DialogueLine = current_dialogue.lines[current_line_index]
    if line.on_complete != "":
        _execute_action(line.on_complete)
    current_line_index += 1
    _show_line()

func _on_choice_selected(index: int, choice: DialogueChoice) -> void:
    choice_made.emit(index)
    if choice.on_select != "":
        _execute_action(choice.on_select)
    if choice.next_line_index >= 0:
        current_line_index = choice.next_line_index
    else:
        _close_dialogue()
        return
    _show_line()

func _close_dialogue() -> void:
    is_active = false
    panel.visible = false
    dialogue_finished.emit()

func _check_condition(condition: String) -> bool:
    # Делегируем проверку условий C++ QuestSystem
    # Формат: "has_item:forest_key", "quest_state:forest_clear:completed"
    var parts := condition.split(":")
    if parts.size() < 2:
        return true
    match parts[0]:
        "has_item":
            return _get_inventory().has_item(StringName(parts[1]))
        "quest_state":
            if parts.size() >= 3:
                return _get_quest_system().get_quest_state(StringName(parts[1])) == _str_to_state(parts[2])
        "has_gold":
            return _get_economy().has_gold(int(parts[1]))
    return true

func _execute_action(action: String) -> void:
    var parts := action.split(":")
    if parts.is_empty():
        return
    match parts[0]:
        "start_quest":
            if parts.size() >= 2:
                _get_quest_system().start_quest(StringName(parts[1]))
        "give_item":
            if parts.size() >= 3:
                _get_inventory().add_item(StringName(parts[1]), int(parts[2]))
        "give_gold":
            if parts.size() >= 2:
                _get_economy().add_gold(int(parts[1]))

func _str_to_state(s: String) -> int:
    match s:
        "not_started": return 0
        "in_progress": return 1
        "completed": return 2
        "failed": return 3
    return -1

func _get_inventory():
    return get_node_or_null("/root/InventorySystem")

func _get_quest_system():
    return get_node_or_null("/root/QuestSystem")

func _get_economy():
    return get_node_or_null("/root/EconomyManager")
```

**Проверка:** ✅ `show_dialogue()` проверяет пустой массив lines. ✅ `_show_line()` проверяет выход за границы → закрытие. ✅ `_check_condition()` — парсинг строки условий, делегирование C++. ✅ `_execute_action()` — парсинг действий (start_quest, give_item, give_gold). ✅ `_update_choices()` очищает старые кнопки. ✅ `is_active` — предотвращает множественные диалоги. ✅ `_on_choice_selected()` — переход к указанной линии или закрытие.

extends Node

const SAVE_PATH := "user://save_game.json"

func save_game() -> bool:
    var save_data := {}

    # Собираем данные от всех менеджеров
    var economy := get_node_or_null("/root/EconomyManager")
    if economy:
        save_data["economy"] = economy.get_save_data()

    var player := get_node_or_null("/root/GameManager").player
    if player:
        var health := player.get_node_or_null("HealthComponent")
        if health:
            save_data["health"] = health.get_save_data()
        var stamina := player.get_node_or_null("StaminaComponent")
        if stamina:
            save_data["stamina"] = stamina.get_save_data()
        save_data["player_position"] = {
            "x": player.global_position.x,
            "y": player.global_position.y,
            "z": player.global_position.z
        }

    # InventorySystem (C++) — через to_dict()
    var inventory := get_node_or_null("/root/InventorySystem")
    if inventory:
        save_data["inventory"] = inventory.to_dict()

    # QuestSystem (C++) — через to_dict()
    var quest_system := get_node_or_null("/root/QuestSystem")
    if quest_system:
        save_data["quests"] = quest_system.to_dict()

    var file := FileAccess.open(SAVE_PATH, FileAccess.WRITE)
    if not file:
        push_error("SaveManager: failed to open save file")
        return false
    file.store_string(JSON.stringify(save_data, "\t"))
    file.close()
    return true

func load_game() -> bool:
    if not FileAccess.file_exists(SAVE_PATH):
        return false

    var file := FileAccess.open(SAVE_PATH, FileAccess.READ)
    if not file:
        return false

    var json := JSON.new()
    var err := json.parse(file.get_as_text())
    file.close()
    if err != OK:
        push_error("SaveManager: failed to parse save file")
        return false

    var save_data: Dictionary = json.data

    # Восстанавливаем данные
    var economy := get_node_or_null("/root/EconomyManager")
    if economy and save_data.has("economy"):
        economy.load_save_data(save_data["economy"])

    var player := get_node_or_null("/root/GameManager").player
    if player and save_data.has("player_position"):
        var pos: Dictionary = save_data["player_position"]
        player.global_position = Vector3(pos.x, pos.y, pos.z)

    if player:
        var health := player.get_node_or_null("HealthComponent")
        if health and save_data.has("health"):
            health.load_save_data(save_data["health"])
        var stamina := player.get_node_or_null("StaminaComponent")
        if stamina and save_data.has("stamina"):
            stamina.load_save_data(save_data["stamina"])

    var inventory := get_node_or_null("/root/InventorySystem")
    if inventory and save_data.has("inventory"):
        inventory.from_dict(save_data["inventory"])

    var quest_system := get_node_or_null("/root/QuestSystem")
    if quest_system and save_data.has("quests"):
        quest_system.from_dict(save_data["quests"])

    return true

func has_save() -> bool:
    return FileAccess.file_exists(SAVE_PATH)

func delete_save() -> void:
    if has_save():
        DirAccess.remove_absolute(SAVE_PATH)
```

**Проверка:** ✅ `FileAccess.open` — правильный API Godot 4. ✅ `JSON.stringify` с отступами. ✅ `JSON.parse` с проверкой ошибки. ✅ Все данные собираются через `get_save_data()`. ✅ Позиция игрока — Vector3 в Dictionary. ✅ `has_save()` и `delete_save()` — утилиты.
