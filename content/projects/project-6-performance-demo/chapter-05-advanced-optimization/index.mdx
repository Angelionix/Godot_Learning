## 5.2 JobSystem — пул потоков

JobSystem реализует пул рабочих потоков для параллельного выполнения задач. Ключевое ограничение: рабочие потоки **не обращаются к Godot API** — они работают только с чистыми C++ данными (компонентами ECS). Все обращения к Godot происходят в главном потоке.

```cpp
// src/job/job_system.h
#pragma once
#include <vector>
#include <queue>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <functional>
#include <atomic>

class JobSystem {
    std::vector<std::thread> workers_;
    std::queue<std::function<void()>> jobs_;
    std::mutex queue_mutex_;
    std::condition_variable cv_;
    std::atomic<bool> stopping_{false};
    std::atomic<int> active_jobs_{0};
    int thread_count_;

    void worker_loop() {
        while (true) {
            std::function<void()> job;
            {
                std::unique_lock<std::mutex> lock(queue_mutex_);
                cv_.wait(lock, [this] {
                    return stopping_.load() || !jobs_.empty();
                });
                if (stopping_.load() && jobs_.empty()) return;
                job = std::move(jobs_.front());
                jobs_.pop();
            }
            active_jobs_.fetch_add(1);
            job();
            active_jobs_.fetch_sub(1);
            cv_.notify_all();  // Уведомить wait_all()
        }
    }

public:
    explicit JobSystem(int thread_count = 0) {
        if (thread_count <= 0) {
            // Количество потоков = ядра - 1 (главный поток не учитываем)
            thread_count = static_cast<int>(
                std::thread::hardware_concurrency()) - 1;
        }
        if (thread_count < 1) thread_count = 1;
        thread_count_ = thread_count;

        for (int i = 0; i < thread_count_; ++i) {
            workers_.emplace_back(&JobSystem::worker_loop, this);
        }
    }

    ~JobSystem() {
        stopping_.store(true);
        cv_.notify_all();
        for (auto& worker : workers_) {
            if (worker.joinable()) worker.join();
        }
    }

    JobSystem(const JobSystem&) = delete;
    JobSystem& operator=(const JobSystem&) = delete;

    // Отправить одну задачу
    void submit(std::function<void()> job) {
        {
            std::lock_guard<std::mutex> lock(queue_mutex_);
            jobs_.push(std::move(job));
        }
        cv_.notify_one();
    }

    // Пакетная отправка: разбивает total_count на батчи по batch_size
    void submit_batch(int total_count, int batch_size,
                      std::function<void(int start, int end)> func) {
        int batches = (total_count + batch_size - 1) / batch_size;
        for (int b = 0; b < batches; ++b) {
            int start = b * batch_size;
            int end = std::min(start + batch_size, total_count);
            submit([func, start, end]() { func(start, end); });
        }
    }

    // Ожидание завершения всех задач
    void wait_all() {
        std::unique_lock<std::mutex> lock(queue_mutex_);
        cv_.wait(lock, [this] {
            return jobs_.empty() && active_jobs_.load() == 0;
        });
    }

    int thread_count() const { return thread_count_; }
};
```

### Параллельное обновление систем

Movement и Boid фазы — самые тяжёлые в кадре. Мы разбиваем их на батчи и выполняем параллельно:

```cpp
// Фрагмент ecs_world.cpp — метод _process

void ECSWorld::_process(double delta) {
    if (!registry_ || !job_system_) return;

    int total = static_cast<int>(registry_->alive_count());
    if (total == 0) return;

    int batch = std::max(1, total / job_system_->thread_count());

    // Фаза 1: параллельное обновление позиций
    job_system_->submit_batch(total, batch,
        [this, delta](int start, int end) {
            movement_system_.update_range(*registry_, delta,
                static_cast<size_t>(start),
                static_cast<size_t>(end));
        });
    job_system_->wait_all();

    // Фаза 2: boids (главный поток — SpatialHash не потокобезопасен)
    boid_system_.update(*registry_, delta);

    // Фаза 3: боевые столкновения
    combat_system_.update(*registry_, delta, spatial_hash_);

    // Фаза 4: culling + LOD (главный поток — нужен Camera3D)
    frustum_culling_.update(*registry_, camera_);
    lod_system_.update(*registry_, camera_);

    // Фаза 5: синхронизация с Godot (главный поток)
    render_system_.update(*registry_);
}
```

## 5.3 Compute-шейдеры

Compute-шейдеры выполняются на GPU и позволяют перенести вычисления с CPU на видеокарту. Это самая мощная одиночная оптимизация в нашем проекте: GPU имеет тысячи ядер, идеально подходящих для параллельной обработки данных.

### GPU-симуляция частиц

```glsl
// shaders/compute_particles.glsl
#[compute]
#version 450

layout(local_size_x = 256) in;

layout(set = 0, binding = 0) buffer ParticleData {
    vec4 positions[];     // xyz = позиция, w = оставшееся время жизни
};

layout(set = 0, binding = 1) buffer ParticleVelocity {
    vec4 velocities[];    // xyz = скорость, w = случайное зерно
};

uniform float delta;
uniform vec3 gravity = vec3(0.0, -9.8, 0.0);

void main() {
    uint idx = gl_GlobalInvocationID.x;
    if (idx >= positions.length()) return;

    // Обновление скорости (гравитация)
    velocities[idx].xyz += gravity * delta;

    // Обновление позиции
    positions[idx].xyz += velocities[idx].xyz * delta;

    // Убывание жизни
    positions[idx].w -= delta;

    // Пересоздание частицы при истечении жизни
    if (positions[idx].w <= 0.0) {
        positions[idx] = vec4(0.0, 5.0, 0.0, 3.0);
        float seed = velocities[idx].w;
        // Псевдослучайное начальное ускорение
        velocities[idx].xyz = vec3(
            (fract(sin(idx + seed) * 43758.5453) - 0.5) * 4.0,
            3.0 + fract(cos(idx + seed) * 23421.631) * 2.0,
            (fract(sin(idx * 1.7 + seed) * 93758.5453) - 0.5) * 4.0
        );
    }
}
```

### GPU-boids: симуляция стаи на GPU

```glsl
// shaders/compute_boids.glsl
#[compute]
#version 450

layout(local_size_x = 256) in;

layout(set = 0, binding = 0) buffer BoidData {
    vec4 boids[];       // xyz = позиция, w = угол направления
};

layout(set = 0, binding = 1) buffer BoidVel {
    vec4 velocities[];  // xyz = скорость, w = максимальная скорость
};

uniform float delta;
uniform float perception_radius = 5.0;
uniform float separation_radius = 1.5;
uniform float separation_weight = 1.5;
uniform float alignment_weight = 1.0;
uniform float cohesion_weight = 0.8;

void main() {
    uint idx = gl_GlobalInvocationID.x;
    if (idx >= boids.length()) return;

    vec3 my_pos = boids[idx].xyz;
    float max_speed = velocities[idx].w;

    vec3 sep = vec3(0.0);
    vec3 ali = vec3(0.0);
    vec3 coh = vec3(0.0);
    int neighbors = 0;

    // Поиск соседей — O(n²) на GPU
    // Для продакшена замените на spatial hashing pass
    for (uint j = 0; j < boids.length(); j++) {
        if (j == idx) continue;

        vec3 diff = my_pos - boids[j].xyz;
        float dist = length(diff);

        if (dist < perception_radius) {
            // Separation — отталкивание
            if (dist < separation_radius && dist > 0.001) {
                sep += normalize(diff) / dist;
            }
            // Alignment — средняя скорость соседей
            ali += velocities[j].xyz;
            // Cohesion — центр масс соседей
            coh += boids[j].xyz;
            neighbors++;
        }
    }

    if (neighbors == 0) return;

    float inv_n = 1.0 / float(neighbors);

    // Итоговый вектор руления с весами
    vec3 steer = vec3(0.0);
    steer += sep * separation_weight;
    steer += normalize(ali * inv_n) * alignment_weight;
    steer += normalize(coh * inv_n - my_pos) * cohesion_weight;

    // Применить руление к скорости
    velocities[idx].xyz += steer * delta;

    // Ограничить скорость
    float speed = length(velocities[idx].xyz);
    if (speed > max_speed) {
        velocities[idx].xyz = normalize(velocities[idx].xyz) * max_speed;
    }

    // Обновить позицию
    boids[idx].xyz += velocities[idx].xyz * delta;
}
```

## 5.4 FrustumCulling

Godot автоматически выполняет frustum culling для узлов дерева, но экземпляры MultiMesh — это один узел. Мы реализуем собственный frustum culling на C++, чтобы скрыть невидимые юниты.

```cpp
// src/utils/frustum.h
#pragma once
#include <godot_cpp/variant/plane.hpp>
#include <godot_cpp/variant/vector3.hpp>
#include <godot_cpp/variant/projection.hpp>
#include <godot_cpp/classes/camera3d.hpp>
#include "../ecs/core.h"
#include "../ecs/components.h"

using namespace godot;

struct Frustum {
    Plane planes[6];  // left, right, top, bottom, near, far

    static Frustum from_camera(Camera3D* cam) {
        Frustum f;
        Projection proj = cam->get_projection();
        Transform3D transform = cam->get_global_transform();
        Projection vp = proj * Projection(transform.inverse());
        Array planes_arr = vp.get_projection_planes();
        for (int i = 0; i < 6 && i < planes_arr.size(); ++i) {
            f.planes[i] = planes_arr[i];
        }
        return f;
    }

    // Проверка: находится ли точка внутри frustum (с радиусом)
    bool contains(const Vector3& point, float radius = 0.0f) const {
        for (int i = 0; i < 6; ++i) {
            if (planes[i].distance_to(point) < -radius) {
                return false;
            }
        }
        return true;
    }
};

class FrustumCullingSystem {
public:
    void update(Registry& reg, Camera3D* cam) {
        if (!cam) return;
        Frustum frustum = Frustum::from_camera(cam);

        reg.view<TransformComponent, RenderableComponent>(
            [frustum](TransformComponent& transform, RenderableComponent& renderable) {
                Vector3 pos(transform.position[0],
                           transform.position[1],
                           transform.position[2]);
                // Помечаем невидимые юниты
                renderable.visible = frustum.contains(pos, 1.0f);
            });
    }
};
```

Frustum culling уменьшает количество видимых экземпляров в MultiMesh. При камере с FOV 60° обычно видно ~30-40% объектов, что снижает нагрузку на GPU.

## 5.5 LODSystem — уровни детализации

LODSystem переключает уровень детализации меша в зависимости от расстояния до камеры. Для 10 000 юнитов большинство находятся далеко от камеры и могут использовать упрощённый меш.

```cpp
// src/utils/lod_system.h
#pragma once
#include "../ecs/core.h"
#include "../ecs/components.h"
#include <godot_cpp/classes/camera3d.hpp>
#include <godot_cpp/variant/vector3.hpp>

using namespace godot;

class LODSystem {
    float lod_distances_[3] = {20.0f, 60.0f, 200.0f};
    int update_interval_ = 4;  // Обновлять LOD каждые 4 кадра
    int frame_counter_ = 0;

public:
    void set_lod_distances(float near, float mid, float far) {
        lod_distances_[0] = near;
        lod_distances_[1] = mid;
        lod_distances_[2] = far;
    }

    void update(Registry& reg, Camera3D* cam) {
        if (!cam) return;

        // Frame-skipping: обновляем LOD не каждый кадр
        ++frame_counter_;
        if (frame_counter_ < update_interval_) return;
        frame_counter_ = 0;

        Vector3 cam_pos = cam->get_global_position();

        reg.view<TransformComponent, RenderableComponent>(
            [cam_pos, this](TransformComponent& transform,
                           RenderableComponent& renderable) {
                float dx = transform.position[0] - cam_pos.x;
                float dz = transform.position[2] - cam_pos.z;
                float dist_sq = dx * dx + dz * dz;

                // Выбираем уровень детализации по дистанции
                // LOD0 — полный меш (500 tri), LOD1 — упрощённый (100 tri),
                // LOD2 — биллборд (2 tri)
                if (dist_sq < lod_distances_[0] * lod_distances_[0]) {
                    renderable.material_index = 0;  // LOD0
                } else if (dist_sq < lod_distances_[1] * lod_distances_[1]) {
                    renderable.material_index = 1;  // LOD1
                } else {
                    renderable.material_index = 2;  // LOD2
                }
            });
    }
};
```

### Frame-skipping

LOD не нужно обновлять каждый кадр — визуальная разница при переключении между кадрами незаметна. Обновляя LOD каждые 4 кадра, мы снижаем стоимость системы в 4×.

| Параметр | LOD0 (близко) | LOD1 (средне) | LOD2 (далеко) |
| --- | --- | --- | --- |
| Дистанция | < 20м | 20-60м | > 60м |
| Треугольников | 500 | 100 | 2 (биллборд) |
| % юнитов (typical) | ~10% | ~30% | ~60% |

## 5.6 PoolAllocator — кастомное выделение памяти

Стандартный `new`/`malloc` — общий назначение и не оптимизирован для паттернов выделения в ECS. PoolAllocator выделяет память блоками фиксированного размера, устраняя фрагментацию и уменьшая накладные расходы.

```cpp
// src/utils/pool_allocator.h
#pragma once
#include <vector>
#include <cstddef>
#include <cassert>

class PoolAllocator {
    std::vector<uint8_t> buffer_;     // Непрерывный буфер памяти
    std::vector<void*> free_list_;    // Список свободных блоков
    size_t block_size_;               // Размер одного блока
    size_t capacity_;                 // Максимальное количество блоков

public:
    PoolAllocator(size_t block_size, size_t capacity)
        : block_size_(block_size), capacity_(capacity) {
        // Выделяем весь буфер сразу
        buffer_.resize(block_size_ * capacity_);
        // Заполняем free list
        free_list_.reserve(capacity);
        for (size_t i = 0; i < capacity; ++i) {
            free_list_.push_back(&buffer_[i * block_size_]);
        }
    }

    // Быстрое выделение — O(1), нет системного вызова
    void* allocate() {
        if (free_list_.empty()) return nullptr;
        void* ptr = free_list_.back();
        free_list_.pop_back();
        return ptr;
    }

    // Быстрое освобождение — O(1)
    void deallocate(void* ptr) {
        assert(ptr >= buffer_.data() &&
               ptr < buffer_.data() + buffer_.size());
        free_list_.push_back(ptr);
    }

    size_t free_count() const { return free_list_.size(); }
    size_t used_count() const { return capacity_ - free_list_.size(); }
};
```

В ECS PoolAllocator используется для быстрого создания/уничтожения сущностей: каждая сущность — это фиксированный блок данных, и аллокация/деаллокация происходит за O(1) без системных вызовов.

<Callout type="warning" title="Закон Амдала">
Закон Амдала ограничивает ускорение от параллелизации долей последовательного кода. Если 10% кода нельзя распараллелить, максимальное ускорение = 1 / (0.1 + 0.9/n) ≈ 10× при бесконечном числе потоков. В нашем случае последовательная доля (Render + Culling + sync) ≈ 25%, что ограничивает теоретическое ускорение до ~4×. На практике мы получаем ~2.3× (40→92 FPS) из-за накладных расходов на синхронизацию.
</Callout>
